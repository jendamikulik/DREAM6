\section{Exact zero-repair problem}
% ============================================================

Let
\[
\mu_P=\sum_xP(x)\delta_x,
\qquad
\mu_Q=\sum_yQ(y)\delta_y.
\]
A depth-\(n\) exact zero-repair tree is a family
\[
\{B_h\}_{|h|\le n}
\]
of probability laws on \(\Z_{\ge0}\), indexed by source histories, satisfying
at each internal node
\begin{equation}
\boxed{
\mu_Q*B_h
=
\sum_xP(x)\,\delta_x*B_{hx}.
}
\tag{ZR}
\end{equation}

The finite-horizon bill is
\[
\boxed{
\Bcal_n
=
\inf\left\{
\E_{B_\varnothing}J:
\{B_h\}_{|h|\le n}\text{ satisfies \emph{(ZR)}}
\right\}.
}
\tag{3}
\]

For the quartic pair,
\[
G_P(z)-G_Q(z)
=
\boxed{\frac{(1-z)^4}{64}},
\tag{4}
\]
and both laws have mean \(2\), variance \(3/2\), and vanishing odd centered
moments.

\begin{proposition}[Monotonicity]
\[
\boxed{
\Kcal_{n+1}\subseteq\Kcal_n,
\qquad
\Bcal_n\le\Bcal_{n+1}.
}
\tag{5}
\]
\end{proposition}

\begin{proof}
Truncate any feasible depth-\((n+1)\) tree after depth \(n\).
\end{proof}

\section{Convexity and stochastic upward closure}
% ============================================================

Every \(\Kcal_n\) is convex by linearity of \emph{(ZR)}.

Write
\[
\widetilde B\succeq_{\st}B
\]
for first-order stochastic dominance.

\begin{theorem}[Upward closure]
If
\[
B\in\Kcal_n,
\qquad
\widetilde B\succeq_{\st}B,
\]
then
\[
\boxed{\widetilde B\in\Kcal_n.}
\tag{8}
\]
\end{theorem}

\begin{proof}
Induct on \(n\).
Couple \(J\sim B\) and \(\widetilde J\sim\widetilde B\) so that
\(\widetilde J\ge J\) almost surely.
Let \(Y\sim Q\) be independent.
Use the existing feasible kernel from \(Y+J\) to the source symbol \(X\sim P\).
Then
\[
Y+\widetilde J\ge Y+J\ge X.
\]
Conditional on \(X=x\), the new residual stochastically dominates the old
child residual, which is viable to depth \(n-1\).
The induction hypothesis completes the construction.
\end{proof}

\tag{32}

near-identity limit and is numerically better conditioned than pushing
\(a\to0\) directly.
